func importFixture(t *testing.T) (store.Session, string) {
	home := t.TempDir()
	t.Setenv("HOME", home)
	t.Setenv("XDG_DATA_HOME", filepath.Join(home, "data"))
	t.Setenv("XDG_STATE_HOME", filepath.Join(home, "state"))
	t.Setenv("XDG_CONFIG_HOME", filepath.Join(home, "config"))
	const cid = "22222222-2222-4222-8222-222222222222"
	s := store.Session{ID: "c1", RootID: "r", Agent: "codex", ClaudeID: cid, Dir: t.TempDir()}
	t.Setenv("CODEX_HOME", codexcfg.AgentHome(s.Dir))
	dir := filepath.Join(codexcfg.AgentHome(s.Dir), "sessions", "2026", "07", "02")
	_ = os.MkdirAll(dir, 0o755)
	roll := filepath.Join(dir, "rollout-2026-07-02T10-00-00-"+cid+".jsonl")
	body := `{"timestamp":"2026-07-02T10:00:00Z","type":"session_meta","payload":{"id":"` + cid + `"}}` + "\n" +
		`{"timestamp":"2026-07-02T10:00:01Z","type":"response_item","payload":{"type":"message","role":"user","content":[{"type":"input_text","text":"PRE-MIGRATION PROMPT"}]}}` + "\n"
	if err := os.WriteFile(roll, []byte(body), 0o644); err != nil {
		t.Fatal(err)
	}
	return s, roll
}

func logText3(t *testing.T, id string) string {
	b, _ := os.ReadFile(codexapp.EventLogPathFor(id))
	return string(b)
}

func newImportDaemon(t *testing.T) *Daemon {
	ctx, cancel := context.WithCancel(context.Background())
	t.Cleanup(cancel)
	d := &Daemon{codex: codexapp.NewManager(ctx, "")}
	t.Cleanup(d.codex.Shutdown)
	return d
}

func TestProbe3MarkerFailureRetry(t *testing.T) {
	s, _ := importFixture(t)
	d := newImportDaemon(t)
	block := historyMarkerPath(s.ID) + ".tmp"
	_ = os.MkdirAll(block, 0o700)
	err1 := d.importHistory(context.Background(), s)
	_ = os.RemoveAll(block)
	err2 := d.importHistory(context.Background(), s)
	_, mErr := os.Stat(historyMarkerPath(s.ID))
	t.Logf("(a) first err=%v second err=%v prefixCopies=%d markerWritten=%v", err1 != nil, err2, strings.Count(logText3(t, s.ID), "PRE-MIGRATION"), mErr == nil)
}

func TestProbe3IdentityWithoutLog(t *testing.T) {
	s, _ := importFixture(t)
	d := newImportDaemon(t)
	_ = codexapp.SaveIdentity(codexapp.Identity{SessionID: s.ID, ThreadID: s.ClaudeID})

}

func TestProbe3IdentityNoticesOnly(t *testing.T) {
	s, _ := importFixture(t)
	d := newImportDaemon(t)
	_ = codexapp.SaveIdentity(codexapp.Identity{SessionID: s.ID, ThreadID: s.ClaudeID, Resumable: true, Version: 1})
	_ = codexapp.AppendNotice(s.ID, "info", "starting agent")
	err := d.importHistory(context.Background(), s)
	txt := logText3(t, s.ID)
	t.Logf("(b2) identity + notices-only log: err=%v historyPresent=%v noticeKept=%v historyBeforeNotice=%v", err, strings.Contains(txt, "PRE-MIGRATION"), strings.Contains(txt, "starting agent"), strings.Index(txt, "PRE-MIGRATION") < strings.Index(txt, "starting agent"))
}

func TestProbe3Concurrent(t *testing.T) {
	bad := 0
	for i := 0; i < 50; i++ {
		t.Run("", func(t *testing.T) {
			s, _ := importFixture(t)
			d := newImportDaemon(t)
			var wg sync.WaitGroup
			for j := 0; j < 3; j++ {
				wg.Add(1)
				go func() { defer wg.Done(); _ = d.importHistory(context.Background(), s) }()
			}
			wg.Wait()
			if strings.Count(logText3(t, s.ID), "PRE-MIGRATION") != 1 {
				bad++
			}
		})
	}
	t.Logf("(c) concurrent d.importHistory x3: %d/50 bad", bad)
}

func TestProbe3NoticesRaceImport(t *testing.T) {
	lost := 0
	for i := 0; i < 50; i++ {
		t.Run("", func(t *testing.T) {
			s, _ := importFixture(t)
			d := newImportDaemon(t)
			var wg sync.WaitGroup
			wg.Add(2)
			go func() { defer wg.Done(); _ = d.importHistory(context.Background(), s) }()
			go func() {
				defer wg.Done()
				for k := 0; k < 20; k++ {
					_ = codexapp.AppendNotice(s.ID, "info", "N"+string(rune('a'+k)))
				}
			}()
			wg.Wait()
			txt := logText3(t, s.ID)
			for k := 0; k < 20; k++ {
				if !strings.Contains(txt, "\"N"+string(rune('a'+k))+"\"") && !strings.Contains(txt, "N"+string(rune('a'+k))) {
					lost++
				}
			}
			if strings.Count(txt, "PRE-MIGRATION") != 1 {
				lost += 100
			}
		})
	}
	t.Logf("(d) AppendNotice x20 racing import, 50 trials: lost notices or bad prefix score=%d", lost)
}
