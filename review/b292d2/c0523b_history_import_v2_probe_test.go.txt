func importFixture(t *testing.T) (store.Session, string) {
	home := t.TempDir()
	t.Setenv("HOME", home)
	t.Setenv("XDG_DATA_HOME", filepath.Join(home, "data"))
	t.Setenv("XDG_STATE_HOME", filepath.Join(home, "state"))
	t.Setenv("XDG_CONFIG_HOME", filepath.Join(home, "config"))
	const cid = "22222222-2222-4222-8222-222222222222"
	s := store.Session{ID: "c1", RootID: "r", Agent: "codex", ClaudeID: cid, Dir: t.TempDir()}
	t.Setenv("CODEX_HOME", codexcfg.AgentHome(s.Dir))
	dir := filepath.Join(codexcfg.AgentHome(s.Dir), "sessions", "2026", "07", "02")
	_ = os.MkdirAll(dir, 0o755)
	roll := filepath.Join(dir, "rollout-2026-07-02T10-00-00-"+cid+".jsonl")
	body := `{"timestamp":"2026-07-02T10:00:00Z","type":"session_meta","payload":{"id":"` + cid + `"}}` + "\n" +
		`{"timestamp":"2026-07-02T10:00:01Z","type":"response_item","payload":{"type":"message","role":"user","content":[{"type":"input_text","text":"PRE-MIGRATION PROMPT"}]}}` + "\n"
	if err := os.WriteFile(roll, []byte(body), 0o644); err != nil {
		t.Fatal(err)
	}
	return s, roll
}

func logText(t *testing.T, id string) string {
	b, _ := os.ReadFile(codexapp.EventLogPathFor(id))
	return string(b)
}

// (a) Marker write fails after the log rename; the retry imports again.
func TestProbe2MarkerFailureDuplicatesPrefix(t *testing.T) {
	s, _ := importFixture(t)
	block := historyMarkerPath(s.ID) + ".tmp"
	_ = os.MkdirAll(block, 0o700) // WriteFile(tmp) fails
	_, err1 := importCodexHistory(s)
	_ = os.RemoveAll(block)
	_, err2 := importCodexHistory(s)
	t.Logf("first err=%v second err=%v prefixCopies=%d", err1 != nil, err2, strings.Count(logText(t, s.ID), "PRE-MIGRATION"))
}

// (b) v0 identity (legacy structured era, no Resumable) + rollout, no log.
func TestProbe2IdentityWithoutLogSkips(t *testing.T) {
	s, _ := importFixture(t)
	if err := codexapp.SaveIdentity(codexapp.Identity{SessionID: s.ID, ThreadID: s.ClaudeID}); err != nil {
		t.Fatal(err)
	}
	imported, err := importCodexHistory(s)
	m, _ := os.ReadFile(historyMarkerPath(s.ID))
	t.Logf("v0 identity, no log: imported=%v err=%v historyPresent=%v marker=%s", imported, err, strings.Contains(logText(t, s.ID), "PRE-MIGRATION"), m)
}

// (c) Two concurrent first imports of one session.
func TestProbe2ConcurrentImports(t *testing.T) {
	dup := 0
	for i := 0; i < 50; i++ {
		t.Run("", func(t *testing.T) {
			s, _ := importFixture(t)
			var wg sync.WaitGroup
			for j := 0; j < 2; j++ {
				wg.Add(1)
				go func() { defer wg.Done(); _, _ = importCodexHistory(s) }()
			}
			wg.Wait()
			if strings.Count(logText(t, s.ID), "PRE-MIGRATION") != 1 {
				dup++
			}
		})
	}
	t.Logf("concurrent imports: %d/50 trials with prefix count != 1", dup)
}

// (d) A writer holding the log open across the replacement (supervisor logW shape).
func TestProbe2HeldWriterAcrossRename(t *testing.T) {
	s, _ := importFixture(t)
	_ = codexapp.AppendNotice(s.ID, "info", "starting agent")
	held, err := os.OpenFile(codexapp.EventLogPathFor(s.ID), os.O_WRONLY|os.O_APPEND, 0o600)
	if err != nil {
		t.Fatal(err)
	}
	_, _ = importCodexHistory(s)
	_, _ = held.Write([]byte("{\"type\":\"notice\",\"payload\":{\"text\":\"HELD-WRITER-AFTER\"}}\n"))
	held.Close()
	txt := logText(t, s.ID)
	t.Logf("held writer: noticeKept=%v historyPresent=%v heldWriteVisible=%v", strings.Contains(txt, "starting agent"), strings.Contains(txt, "PRE-MIGRATION"), strings.Contains(txt, "HELD-WRITER-AFTER"))
}
